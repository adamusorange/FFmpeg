# android の構成の configure の引数のうち、部品の選び方。1 行に 1 つ以上、# から行末は読まない。
# 向け先とツールチェーンの引数 (--target-os=android、NDK の clang など) は build.sh が足す。
# slim/configure.txt と同じ選び方で、D3D11VA・NVENC と変換の部品を持たない。
# 映像のハードウェアのデコードは、FFmpeg を通さずに MediaCodec を使う (DamukoPla の plan 10.8)。

# ライセンスと出力の形は slim と同じ (LGPL の .so)
--enable-version3 --disable-debug --enable-shared --disable-static
# configure の警告 (did not match anything、Disabled ... because) で止める
--fatal-warnings

# 自動で見つけたものは入れない。要るものは下で明示する (pthreads も自動で見つけるものに入る)
--disable-autodetect
--enable-pthreads
--enable-zlib --enable-libdav1d --enable-libsoxr

# ライブラリだけを使う
--disable-programs --disable-doc
--disable-avdevice --disable-devices --disable-network

# mpeg4 と mp4 は単体テストが見本を作るのに使う
--disable-encoders --enable-encoder=mpeg4
--disable-muxers --enable-muxer=mp4
# 再生は自前の AVIOContext で読む。file は単体テストが見本を読み書きするのに使う
--disable-protocols --enable-protocol=file

# abuffer / abuffersink は常に入り、ここでは選べない。
# 音声は atempo (速度)・dynaudnorm (音量)・aresample・aformat、映像は format・scale (ソフトウェアで解いたコマを出すとき)
--disable-filters --enable-filter=atempo,dynaudnorm,aresample,aformat,format,scale

--disable-hwaccels
